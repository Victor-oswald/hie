cmake_minimum_required(VERSION 3.16)
project(hie VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(HIE_NATIVE        "Tune for the build machine (-march=native); disables runtime dispatch benefit on other CPUs" OFF)
option(HIE_BUILD_SHARED  "Also build libhie_c (shared, C API only)" ON)
option(HIE_BUILD_TOOLS   "Build hie_cli" ON)
option(HIE_BUILD_TESTS   "Build tests" ON)
option(HIE_BUILD_V4L2    "Build hie_webcam (Linux V4L2 capture + v4l2loopback output)" ON)
option(HIE_WIN_AVX2       "Windows: build for AVX2 CPUs (2013+). OFF keeps old CPUs working" OFF)
option(HIE_SANITIZE      "ASan+UBSan (debug testing)" OFF)

find_package(Threads REQUIRED)
if(MINGW)
  set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -static")
endif()

set(HIE_SOURCES src/enhancer.cpp src/kernels.cpp src/lut3d.cpp src/colorpipe.cpp src/analysis.cpp src/capi.cpp)

add_library(hie_obj OBJECT ${HIE_SOURCES})
target_include_directories(hie_obj PUBLIC include PRIVATE src)
set_target_properties(hie_obj PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(MSVC)
  target_compile_options(hie_obj PRIVATE /O2 /W3)
  if(HIE_WIN_AVX2)
    target_compile_options(hie_obj PRIVATE /arch:AVX2)
  endif()
else()
  target_compile_options(hie_obj PRIVATE -O3 -Wall -Wextra -Wno-unused-parameter)
  if(HIE_NATIVE)
    target_compile_options(hie_obj PRIVATE -march=native)
  endif()
  if(HIE_WIN_AVX2 AND WIN32)
    target_compile_options(hie_obj PRIVATE -mavx2)
  endif()
  if(HIE_SANITIZE)
    target_compile_options(hie_obj PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -O1)
  endif()
endif()

add_library(hie STATIC $<TARGET_OBJECTS:hie_obj>)
target_include_directories(hie PUBLIC include)
target_link_libraries(hie PUBLIC Threads::Threads)
if(HIE_SANITIZE AND NOT MSVC)
  target_link_options(hie PUBLIC -fsanitize=address,undefined)
endif()

if(HIE_BUILD_SHARED)
  add_library(hie_c SHARED $<TARGET_OBJECTS:hie_obj>)
  target_include_directories(hie_c PUBLIC include)
  target_link_libraries(hie_c PRIVATE Threads::Threads)
  if(NOT MSVC)
    target_compile_options(hie_c PRIVATE -fvisibility=hidden)
  endif()
endif()

if(HIE_BUILD_TOOLS)
  add_executable(hie_cli tools/hie_cli.cpp)
  target_include_directories(hie_cli PRIVATE third_party)
  target_link_libraries(hie_cli PRIVATE hie)
  if(NOT MSVC)
    target_compile_options(hie_cli PRIVATE -O2 -Wall)
  endif()
  include(CheckIncludeFileCXX)
  check_include_file_cxx(linux/videodev2.h HIE_HAVE_V4L2)
  if(HIE_BUILD_V4L2 AND HIE_HAVE_V4L2 AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(hie_webcam tools/hie_webcam.cpp)
    target_link_libraries(hie_webcam PRIVATE hie)
  endif()
endif()

if(HIE_BUILD_TESTS)
  enable_testing()
  add_executable(hie_tests tests/test_main.cpp)
  target_include_directories(hie_tests PRIVATE src third_party)
  target_link_libraries(hie_tests PRIVATE hie)
  if(NOT MSVC)
    target_compile_options(hie_tests PRIVATE -O2 -Wall)
  endif()
  add_test(NAME hie_tests COMMAND hie_tests)
endif()

install(TARGETS hie ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(HIE_BUILD_SHARED)
  install(TARGETS hie_c LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
endif()
